Monthly Revenue Trend GMV + MoM Growth

WITH monthly AS (
  SELECT
    DATE_TRUNC('month', o.order_date) AS month,
    SUM(o.bill_amount)              AS gmv,
    COUNT(*)                         AS order_count
  FROM orders o
  JOIN merchants m ON m.merchant_id = o.merchant_id
  WHERE o.status = 'completed'
  GROUP BY 1
)
SELECT
  month,
  gmv,
  order_count,
  LAG(gmv) OVER (ORDER BY month)  AS prev_gmv,
  ROUND(
    (gmv - LAG(gmv) OVER (ORDER BY month))
    / NULLIF(LAG(gmv) OVER (ORDER BY month), 0)
    * 100, 1
  ) AS mom_growth_pct
FROM monthly
ORDER BY month;

Category Distribution by GMV share

SELECT
  m.category,
  SUM(o.bill_amount) AS total_gmv,
  ROUND(
    SUM(o.bill_amount) * 100.0
    / SUM(SUM(o.bill_amount)) OVER (),
    1
  ) AS pct_share
FROM orders o
JOIN merchants m ON m.merchant_id = o.merchant_id
WHERE o.status = 'completed'
GROUP BY m.category
ORDER BY total_gmv DESC;

Top 10 Merchants by total revenue

Rank Merchant Category City Orders Revenue Avg Order Rating 
SELECT
  RANK() OVER (
    ORDER BY SUM(o.bill_amount) DESC
  ) AS rnk,
  m.name,
  m.category,
  m.city,
  COUNT(o.order_id)              AS total_orders,
  SUM(o.bill_amount)              AS total_revenue,
  ROUND(AVG(o.bill_amount), 0)     AS avg_order_value,
  ROUND(AVG(m.rating), 1)          AS avg_rating
FROM orders o
JOIN merchants m ON m.merchant_id = o.merchant_id
WHERE o.status = 'completed'
GROUP BY m.merchant_id, m.name, m.category, m.city
HAVING COUNT(o.order_id) > 1
ORDER BY total_revenue DESC
LIMIT 10;

City-wise GMV completed orders

SELECT
  u.city,
  COUNT(o.order_id)            AS total_orders,
  SUM(o.bill_amount)            AS total_gmv,
  COUNT(DISTINCT o.user_id)    AS unique_users,
  ROUND(AVG(o.bill_amount), 0)   AS avg_order_value
FROM orders o
JOIN users u ON u.user_id = o.user_id
WHERE o.status = 'completed'
GROUP BY u.city
ORDER BY total_gmv DESC;

Hourly Order Pattern order volume by hour

SELECT
  EXTRACT(HOUR FROM o.order_date) AS hour_of_day,
  COUNT(*)                          AS order_count,
  SUM(o.bill_amount)                 AS hourly_gmv,
  ROUND(AVG(o.bill_amount), 0)        AS avg_bill
FROM orders o
WHERE o.status = 'completed'
GROUP BY 1
ORDER BY 1;

Day-Hour Heatmap order density by day & hour

SELECT
  FORMAT_DATE('%A', o.order_date)    AS day_name,
  EXTRACT(DAYOFWEEK FROM o.order_date) AS dow,
  EXTRACT(HOUR FROM o.order_date)      AS hour_of_day,
  COUNT(*)                             AS order_count
FROM orders o
WHERE o.status = 'completed'
GROUP BY 1, 2, 3
ORDER BY dow, hour_of_day;

Customer Cohort Retention monthly cohorts

WITH first_order AS (
  SELECT
    user_id,
    DATE_TRUNC('month', MIN(order_date)) AS cohort_month
  FROM orders
  WHERE status = 'completed'
  GROUP BY user_id
),
activity AS (
  SELECT
    f.cohort_month,
    DATE_DIFF(
      DATE_TRUNC('month', o.order_date),
      f.cohort_month,
      MONTH
    ) AS month_offset,
    COUNT(DISTINCT o.user_id) AS active_users
  FROM orders o
  JOIN first_order f ON f.user_id = o.user_id
  WHERE o.status = 'completed'
  GROUP BY 1, 2
),
cohort_size AS (
  SELECT cohort_month, COUNT(*) AS total_users
  FROM first_order
  GROUP BY 1
)
SELECT
  a.cohort_month,
  a.month_offset,
  a.active_users,
  c.total_users,
  ROUND(a.active_users * 100.0 / c.total_users, 1) AS retention_pct
FROM activity a
JOIN cohort_size c ON c.cohort_month = a.cohort_month
ORDER BY a.cohort_month, a.month_offset;

Order Funnel conversion stages

SELECT
  COUNT(DISTINCT user_id)                         AS total_users,
  COUNT(DISTINCT CASE
    WHEN status IN ('completed','pending','cancelled','refunded')
    THEN user_id END)                               AS placed_order,
  COUNT(DISTINCT CASE
    WHEN status = 'completed'
    THEN user_id END)                               AS completed_order,
  COUNT(DISTINCT CASE
    WHEN status = 'completed' AND cashback_amount > 0
    THEN user_id END)                               AS earned_cashback,
  COUNT(DISTINCT CASE
    WHEN user_id IN (
      SELECT user_id FROM orders
      WHERE status = 'completed'
      GROUP BY user_id HAVING COUNT(*) >= 2
    )
    THEN user_id END)                               AS repeat_customer
FROM orders;
